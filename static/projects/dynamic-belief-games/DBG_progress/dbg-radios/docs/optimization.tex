\section{Multi-Agent Optimization V2}

\noindent \textbf{Multi-Agent Optimization Problem For Node $i$}
\\
\noindent\fbox{%
    \parbox{\textwidth}{%
        \begin{align*}
            \begin{split}
            \max_{\pi_{k}, \tilde{q}_{k}^{(i)} \in \mathcal{Q}} \sum_{k=0}^{T-1} \sum_{j=1, j \not = i}^{n} [ &\mathbb{E}[\frac{\text{pkts}}{\text{unit time}} | G_{k}, \pi_{k}, H_{\psi_{k}} = 1] - \gamma_{1} \mathbb{E}[\frac{\text{pkts}}{\text{unit time}} | G_{k}, \pi_{k}, H_{\psi_{k}} > 1] \\  &-
            \gamma_{2} \sum_{f \in F} log(p(T_{f})) p(T_{f}) - \gamma_{3} \text{Cost}_{\text{queue}}(\pi_{k}^{(i)}) ]
            \end{split}
        \end{align*}
        \nonumber
    }%
}

\vspace{1 cm}


\textbf{Definitions}
\begin{itemize}
    \item $G_{k} = \{ (V, E_{k}) | E_{k} = d(n_{k}^{(i)}, n_{k}^{(j)}) \le d_{\text{max}} \}$ is the connectivity graph created by the TSM radio software where $V$ is the set of vertices and $E_{k}$ is the set of edges as time $k$.
    \item $\pi_{k} =$ queue priority weights (including drops) at time $k$ $(\pi_{k} \in [0, 1]^{F+1})$
    \item $\tilde{q_{k}}^{(i)}$ is the priority queue for node $i$ at time $k$.
    \item $T_{f} = | \{ \text{pkts} \in f\} |$ is the amount of traffic on a frequency $f$.
    \item $\log(p(T_{f})) p(T_{f})$ is the Shannon entropy of network traffic at frequency $f$.
    \item $\text{Cost}_{\text{queue}}(\pi_{k}^{(i)})$ is the penalty on dropping traffic and reordering the queue.
    \item $\Gamma = (\gamma_{1}, \gamma_{2})$ are hyperparameters to tune.
\end{itemize}


\vspace{1 cm}

\textbf{Explaining Each Term}
\begin{itemize}
    \item $\mathbb{E}[\frac{\text{pkts}}{\text{unit time}} | G_{k}, \pi_{k}]$: Promotes successful packet transmission.
    \item $-\sum_{f \in F} log(p(T_{f})) p(T_{f})$: Promotes uniform communication across frequencies.
    \item $\text{Cost}_{\text{queue}}(\pi_{k}^{(i)})$: Penalize dropping packets or re-ordering queue
\end{itemize}

\section{Reward Function for TSM Radio}

\vspace{1 cm}

\noindent \textbf{Definitions}

\begin{itemize}
    \item Let $Q_{k}^{(i)} = [\psi_{k}^{(1)}, \psi_{k}^{(2)}, \dots, \psi_{k}^{(n)}] \in \mathcal{Q}$ be the priority queue at time $k$ where $\mathcal{Q}$ is the priority queue space.
    \item Let $fe: \Psi \to \mathbb{R}^{C}$ be a function where $fe(\psi_{k}^{(\zeta)})$ are the features of the packet used to construct the state space, $\Psi$ is the packet space, $C$ is the number of features to be considered, and $1 \le \zeta \le n$.
    \item $\mathcal{B}: \Psi \to \mathbb{Z}_{+}$ is the product of the amount of bits a packet has and the number of nodes in $G_{k}$.
    \item $\mathcal{R}: \Psi \to \mathbb{Z}_{+}$ weighs the number of nodes that received the bits of a packet against the number of redundant packet transmissions.
    %\item $\sigma: \mathbb{R} \to \mathbb{R}$ is the softmax function.
    \item $\phi$: is a variational inference function.
    \item $g$: is a smoothing function for differentiability.



\end{itemize}\\

Then, we can define the \textbf{state space}, $\mathcal{S}$ of a given node in the network to be the queue defined by 
$$
[fe(\psi_{k}^{\zeta \in [1, n]})] =
\begin{bmatrix}
    \horzbar & fe(\psi_{k}^{(1)}) & \horzbar \\
    \horzbar & fe(\psi_{k}^{(2)}) & \horzbar \\
    & \vdots    & \\
    \horzbar & fe(\psi_{k}^{(n)}) & \horzbar \\
\end{bmatrix} 
\in \mathbb{R}^{n \times C}
$$\\

Our \textbf{action space}, $\mathcal{A}$, is primarily defined to contain only the actions:

\begin{itemize}
    \item \textbf{Queue Reorder}: Swap packets in the queue (swap rows in $[fe(\psi_{k}^{\zeta \in [1, n]})]$).
    \item \textbf{Packet Dropping}: Drop packets in queue (drop rows in $[fe(\psi_{k}^{\zeta \in [1, n]})]$)
\end{itemize}

\vspace{0.5 cm}

Then, we can state our \textbf{policy optimization} objective as
\\
\noindent\fbox{%
    \parbox{\textwidth}{%
    \begin{equation}
        \max_{Q_{k} \in \mathcal{Q}} R(Q_{k}) = \max_{Q_{k} \in \mathcal{Q}} \sum_{\psi_{k}\in Q_{k}} [ \int_{0}^{k}  \frac{g(\mathcal{R}(\psi_{k}))}{\mathcal{B}(\psi_{k})} dk -  \log(\phi(\psi_{k})) \mathbb{E}_{T}[\psi_{k} | G_{k}] \ ]
    \end{equation}
    }%
}

\section{Multi-Agent Optimization V3}

\noindent \textbf{Multi-Agent Optimization Problem For Node $i$ Using TSM Radio Version 2}
\\
\noindent\fbox{%
    \parbox{\textwidth}{%
        \begin{align*}
            \begin{split}
            \max_{\pi_{t}} \sum_{t=0}^{T-1} [ \mathbb{E}[\frac{\text{card}(\Psi_{t}^{(i)})}{\text{unit time}} | G_{t}, \pi_{t}] + \gamma_{1} \sum_{\psi \in \Psi_{t}} R(\psi; a) +
            \gamma_{2} H(T_{t}(f)) + \gamma_{3} \text{Cost}_{\text{queue}}(\pi_{t}^{(i)}; \lambda) ]
            \end{split}
        \end{align*}
        \nonumber
    }%
}

\vspace{1 cm}


\textbf{Definitions}
\begin{itemize}
    \item $\Psi_{t}^{(i)}$ is the set of packets seen by node $i$ at time $t$.
    \item $G_{t} = \{ (V, E_{t}) | E_{k} = d(n_{t}^{(i)}, n_{t}^{(j)}) \le d_{\text{max}} \}$ is the connectivity graph created by the TSM radio software where $V$ is the set of vertices and $E_{k}$ is the set of edges as time $k$.
    \item $\pi_{t}^{(i)}\in\Delta^{F+1}$ are the queue priority weights for node $i$ at time $k$ (simplex constraint: nonnegative and sum to 1); one entry can represent ``drop''
    % \item $\tilde{q_{k}}^{(i)}$ is the priority queue for node $i$ at time $k$.
    \item $R$ is a reward associated with how many times a packet is seen by a node. $R_(\psi; a) = \exp(- \frac{(\psi - 1)^{2}}{a})$
    \item $T_{t}(f) = | \{ \text{pkts} \in f\} |$ is the amount of traffic on a frequency $f$ at time $t$.
    \item $H(T_{t}(f))= \sum _{f \in F} -\log(p(T_{t}(f))) p(T_{t}(f))$ is the Shannon entropy of network traffic at frequency $f$ and time $t$ where $F$ is the set of available frequencies.
    \item $\text{Cost}_{\text{queue}}(\pi_{t}^{(i)}; \lambda) = \lambda \mathbb{E}[\text{pkt drops}| \Psi_{t}^{(i)}] + (1 - \lambda) \mathbb{E}[\text{queue reorders}| \pi_{t}^{(i)}]$ is the penalty on dropping traffic and reordering the queue.
    \item $\Gamma = (\gamma_{1}, \gamma_{2}, \gamma_{3})$, $\lambda, a \in (0, 1)$ are hyperparameters to tune.
\end{itemize}


\vspace{1 cm}

\textbf{Explaining Each Term}
\begin{itemize}
    \item $ [ \mathbb{E}[\frac{\text{card}(\Psi_{t})}{\text{unit time}} | G_{t}, \pi_{t}]$: Promotes reception of as many unique packets as possible.
    \item $R(\psi; a)$: Penalizes redundant packet transmissions
    \item $H(T_{t}(f))= -\log(p(T_{t}(f))) p(T_{t}(f))$: Promotes uniform communication across frequencies.
    \item $\text{Cost}_{\text{queue}}(\pi_{k}^{(i)})$: Penalize dropping packets or re-ordering queue.
\end{itemize}

\textbf{Limited Scalability}
The first term in the function

$$
[ \mathbb{E}[\frac{\text{card}(\Psi_{t})}{\text{unit time}} | G_{t}, \pi_{t}]
$$
directly rewards an increase in the number of packets in the system. Additionally, the term
$$
(T_{t}(f))= -\log(p(T_{t}(f))) p(T_{t}(f))
$$
prevents packets from clustering on a single frequency.

\textbf{Delay}
The term
$$
\text{Cost}_{\text{Protocol}}(\pi_{t}^{(i)}; \beta) = \beta (\mathbb{E}[ \text{collisions} | \pi_{t}^{(i)}] + \mathbb{E}[ \text{backoff time} | \pi_{t}^{(t)}])
$$
can be leveraged to address delays in communication by penalizing large backoff times that create transmission delays. 

\textbf{Limited Reliability}
The term
$$
(T_{t}(f))= -\log(p(T_{t}(f))) p(T_{t}(f))
$$
addresses limited reliability by allowing packets to disburse across frequencies.

            &\mathbb{E}[\frac{\text{card}(\Psi_{t}^{(i)})}{\text{unit time}} | G_{t}, \pi_{t}] + \sum_{\psi \in \Psi_{t}}[ \gamma_{1} R_{\text{priority}}(\psi) + \gamma_{2} R_{\text{hop}}(\psi)  +  \gamma_{3} R_{\text{essential}}(\psi)]
            \\ &+ \gamma_{4} H(T_{t}(f)) + \gamma_{5} \text{Cost}_{\text{queue}}(\varphi_{t}^{(i)}; \lambda)
    \end{split}
\end{align*}\nonumber


\noindent \textbf{Multi-Agent Optimization Problem For Node $i$}
\\
\noindent\fbox{%
    \parbox{\textwidth}{%
    \centering
    \max_{\pi} \mathbb{E}_{\tau \sim P(\tau | \pi)} [ \int_{0}^{T} r_{t}^{(i)}(\tau) dt]
        \nonumber
    }%
}

\vspace{1 cm}


\textbf{Definitions}
\begin{itemize}
    \item $\Psi_{t}^{(i)}$ is the set of packets seen by node $i$ at time $t$.
    \item $G_{t} = \{ (V, E_{t}) | E_{k} = d(n_{t}^{(i)}, n_{t}^{(j)}) \le d_{\text{max}} \}$ is the connectivity graph created by the radio software where $V$ is the set of vertices and $E_{k}$ is the set of edges as time $k$.
    \item $\pi_{t}^{(i)}\in\Delta^{F+1}$ are the queue priority weights for node $i$ at time $k$ (simplex constraint: nonnegative and sum to 1); one entry can represent ``drop''
    % \item $\tilde{q_{k}}^{(i)}$ is the priority queue for node $i$ at time $k$.
    \item $R_{\text{priority}}$ is a smooth function that promotes moving high priority packets to the top of the queue. $R_{\text{priority}}(\psi) = \exp( 4 - \psi.\text{priority}) / \text{packet order in queue}$


    \item $R_{\text{hop}}$ is a smooth function that promotes moving packets with high hop counts to the top of the queue. $R_{\text{hop}}(\psi) = \log(\psi.\text{hop\_count} + 1) / \text{packet order in queue}$

    \item $R_{\text{essential}}$ is a smooth function that rewards nodes for keeping packets that are not redundant. $R_{\text{essential}}(\psi) = \exp(- (\text{node count for }\psi - 1)^{2})$

    \item $T_{t}(f) = | \{ \text{pkts} \in f\} |$ is the amount of traffic on a frequency $f$ at time $t$.
    \item $H(T_{t}(f))= \sum _{f \in F} -\log(p(T_{t}(f))) p(T_{t}(f))$ is the Shannon entropy of network traffic at frequency $f$ and time $t$ where $F$ is the set of available frequencies.
    \item $\text{Cost}_{\text{queue}}(\pi_{t}^{(i)}; \lambda) = \lambda \mathbb{E}[\text{pkt drops}| \Psi_{t}^{(i)}] + (1 - \lambda) \mathbb{E}[\text{queue reorders}| \pi_{t}^{(i)}]$ is the penalty on dropping traffic and reordering the queue.
    \item $\Gamma = (\gamma_{1}, \gamma_{2}, \gamma_{3}, \gamma_{4}, \gamma_{5})$ are hyperparameters to tune.
\end{itemize}

\begin{itemize}
